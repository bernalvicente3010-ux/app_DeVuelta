import 'dart:typed_data';

import 'package:flutter/material.dart';
import 'package:proy3/constants.dart';

class ObjetoCard extends StatelessWidget {
  final String titulo;
  final String tipo;
  final String categoria;
  final String estado;
  final Uint8List? fotoData;
  final bool esAutor;
  final VoidCallback? onTap;
  final VoidCallback? onMarcarDevuelto;
  final VoidCallback? onBorrar;

  const ObjetoCard({
    super.key,
    required this.titulo, 
    required this.tipo,
    required this.categoria,
    required this.estado,
    this.fotoData,
    this.esAutor = false,
    this.onTap,
    this.onMarcarDevuelto,
    this.onBorrar,
  });

  @override
  Widget build(BuildContext context) {
    final bool esDevuelto = estado == 'Devuelto';

    return Card(
      margin: EdgeInsets.symmetric(horizontal: 12, vertical: 6),
      elevation: 10,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      color: esDevuelto ? Colors.green.shade50 : Colors.white,
      child: ListTile(
        leading: ClipRRect(
          borderRadius: BorderRadius.circular(8),
          child: fotoData != null
              ? Image.memory(
                  fotoData!,
                  width: 60,
                  height: 60,
                  fit: BoxFit.cover,
                )
              : Container(
                  width: 60,
                  height: 60,
                  color: Colors.grey.shade300,
                  child: Icon(Icons.image, size: 30),
                ),
        ),
        title: Text(titulo, style: TextStyle(fontWeight: FontWeight.bold)),
        subtitle: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text('$tipo • $categoria'),
            Container(
              padding: EdgeInsets.symmetric(horizontal: 8, vertical: 4),
              decoration: BoxDecoration(
                color: esDevuelto ? kSuccessColor : kWarningColor,
                borderRadius: BorderRadius.circular(12),
              ),
              child: Text(
                estado,
                style: TextStyle(color: Colors.white, fontSize: 12),
              ),
            ),
          ],
        ),
        trailing: esAutor
            ? PopupMenuButton<String>(
                onSelected: (value) {
                  if (value == 'devuelto') onMarcarDevuelto?.call();
                  if (value == 'borrar') onBorrar?.call();
                },
                itemBuilder: (_) => [
                  PopupMenuItem(
                    value: 'devuelto',
                    child: Text('Marcar devuelto'),
                  ),
                  PopupMenuItem(value: 'borrar', child: Text('Borrar')),
                ],
              )
            : null,
        onTap: onTap,
      ),
    );
  }
}
